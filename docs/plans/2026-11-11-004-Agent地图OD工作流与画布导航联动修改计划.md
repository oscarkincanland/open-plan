**前端（client/src/App.jsx，L2565 `onMapAction` 回调改为先处理再透传）**：

```jsx
onMapAction={(action) => {
  // 创建即交付（2026-11-11-004 §3.1）：agent 新建地图画布后自动绑定当前会话并被动打开地图预览。
  if (action?.action === "project_created" && String(action.project || "")) {
    const sid = currentSessionIdRef.current || "";
    if (sid && (!action.sessionId || action.sessionId === sid)) {
      const project = String(action.project);
      fetch(`/api/sessions/${encodeURIComponent(sid)}/map-context`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ mapProject: project, mapView: null }),
      })
        .then((r) => (r.ok ? r.json() : Promise.reject(new Error(`HTTP ${r.status}`))))
        .then(() => {
          refreshSessions?.(); // 实施时以 App 既有会话刷新函数为准（归档/置顶后刷新同一入口）
          const previewTab = (canvasTabsRef.current || []).find((t) => t.kind === "mapPreview") || null;
          const act = resolveMapPreviewAction({
            session: { id: sid, mapProject: project, mapView: null },
            previewTab,
            userInitiated: false, // 被动：补建/同步，不抢用户正在看的标签
          });
          if (act.type === "open" || act.type === "update") openCanvasView("mapPreview", act.payload);
        })
        .catch(() => { /* 绑定失败不打断 Run；用户仍可在地图项目菜单手动切换 */ });
    }
  }
  mapBridgeRef.current?.onMapAction?.(action);
}}
```

绑定放在前端（POST 既有 map-context 端点），服务端不直接写会话头——与"前端拥有会话状态"现状一致；下一条消息提交时 ChatPanel 自动带新 `task.mapProject`（既有逻辑）。

### 3.3 F1-c 新增 `map_export` 工具

**注册位置**：server/agent.mjs map_project_create 工具定义之后（约 L3927）。

- name：`map_export`；label「地图截图导出」；语义组 map、风险 write。
- 参数：`project`（缺省 `entry.task?.mapProject`）、`format`（png|pdf，默认 png）、`size`（a4l/a4p/square）与可选 `width/height`（自定义，边界同批量导出）、`layers`（可选逗号分隔图层白名单）、`title`（可选图面标题条）。
- 执行：`RUNTIME_FLAGS.mapExport` 关闭时直接返回明确错误（`MAP_EXPORT_DISABLED`，不改开关默认值）→ `requireToolApproval`（ask）→ `tryAcquireExportSlot("agent")`（冲突返回 `MAP_EXPORT_BUSY`）→ 复用 `批量导出.planExportJobs` 做参数校验与文件名生成（单项目单尺寸单格式 = 1 个 job）→ `executeExportJob(job,{workspace, outDir: maps/{project}/agent/exports})` → `map.appendAgentManifest(project,{action:"export",title,paths:[相对路径],bytes,runId,sessionId})` → `emitChannelSafe(entry,"file_changed",{files:[…]})` → 返回产物相对路径与大小；finally `releaseExportSlot("agent")`。
- 实施核对项（不改变设计）：`map.projectDir` 若未导出则用 `map.ensureProject(name)`（已导出）取目录；`planExportJobs` 若不支持单 job layers 传法，则在工具内自行过滤 `job.layers`，不改其签名。

### 3.4 F1-d OD 工作流提示词（server/agent.mjs map 工具组描述）

```js
/**
 * 单板居中适配：返回把 path 对应板放到视口中央所需的 { scale, tx, ty }。
 * 口径与 boardsFitViewport 一致（屏幕 = 世界 * scale + t），scale 上限 1（单板不放大超过 100%）。
 * 找不到板或视口非法返回 null。
 */
export function 定位板视图(boards = [], path = "", viewport = {}, 边距 = 48) {
  const 项 = (Array.isArray(boards) ? boards : []).find((b) => b?.path === path);
  const 宽 = Number(viewport?.w); const 高 = Number(viewport?.h);
  if (!项 || !(宽 > 0) || !(高 > 0)) return null;
  const w = Math.max(Number(项.w) || 1, 1);
  const h = Math.max(Number(项.h) || 1, 1);
  const 边 = Math.max(8, Number(边距) || 48);
  const scale = 数值(Math.min((宽 - 边 * 2) / w, (高 - 边 * 2) / h, 1), 0.05, 1, 1);
  return {
    scale,
    tx: (宽 - w * scale) / 2 - (Number(项.x) || 0) * scale,
    ty: (高 - h * scale) / 2 - (Number(项.y) || 0) * scale,
  };
}

/**
 * 轻量缩略数据：把板元素压成色块列表（导航用，不含文本内容细节）。
 * 返回 { background, blocks:[{x,y,w,h,color,kind}], elementCount }；kind ∈ rect|text|image|svg|other。
 * 元素字段名以 normalizeElement（本文件 L220）为准；超过 120 个元素截断。
 */
export function 板缩略数据(board = null) {
  if (!board || !Array.isArray(board.elements)) return null;
  const blocks = board.elements.slice(0, 120).map((e) => ({
    x: Number(e.x) || 0, y: Number(e.y) || 0,
    w: Math.max(Number(e.w) || 2, 1), h: Math.max(Number(e.h) || 2, 1),
    color: String(e.background || e.color || e.fill || "#6b7280"),
    kind: e.type === "text" ? "text" : e.type === "image" ? "image" : e.type === "svg" ? "svg" : e.type === "line" ? "other" : "rect",
  }));
  return { background: String(board.background || ""), blocks, elementCount: board.elements.length };
}
```
